import type { BrowserContext, Page } from '@playwright/test'
import { expect, expectAccessible, expectNoHorizontalOverflow, test, WIDTHS } from './fixtures'

// Independent QA checks for Stories 8.5, 8.6 (screen 23). Web-only run, /v1 answered by the test.
const EVID = 'docs/qa-handoffs/8-5-8-6-report-and-block-evidence'
const ME = '01990000-0000-7000-8000-0000000000e1'
const HIM = '01990000-0000-7000-8000-0000000000e2'
const CONV = '01990000-0000-7000-8000-0000000000c1'
const HIS = '01990000-0000-7000-8000-000000000002'
const MINE = '01990000-0000-7000-8000-000000000001'
const json = (body: unknown, status = 200) => ({ status, contentType: 'application/json', body: JSON.stringify(body) })
const err = (code: string, message: string, status: number) => json({ error: { code, message, details: null, request_id: 'r', retryable: false } }, status)

type S = { posts: any[]; mode: 'ok' | '429' | '401' | 'abort' | 'slow'; pseudonym: string }
async function serve(context: BrowserContext, baseURL: string, s: S) {
  await context.addCookies([{ name: 'ankanu_session', value: 'web', url: baseURL }])
  for (const host of ['https://cdn.tailwindcss.com/**', 'https://fonts.googleapis.com/**', 'https://fonts.gstatic.com/**']) {
    const tw = host.includes('tailwind')
    await context.route(host, (r) => r.fulfill({ status: 200, contentType: tw ? 'application/javascript' : 'text/css', body: tw ? 'window.tailwind = {}' : '' }))
  }
  await context.route('**/v1/**', async (route) => {
    const req = route.request(); const u = new URL(req.url()); const m = req.method()
    if (u.pathname === '/v1/realtime/socket.io.min.js') return route.fulfill({ status: 200, contentType: 'application/javascript', body: 'window.io = undefined' })
    if (u.pathname === `/v1/profiles/${HIM}`) return route.fulfill(json({ profile: { id: HIM, pseudonym: s.pseudonym, age: 29, city: 'ouagadougou', country: 'BF', profession: null, marital_status: 'single', polygamy_intent: 'no', madhhab: null, practice: null, life_plans: null }, verification: { phone: true, id: true, mahram: false }, photo_blur: true, photos: [] }))
    if (u.pathname === `/v1/conversations/${CONV}` && m === 'GET') return route.fulfill(json({ conversation: { id: CONV, stage: 'chat', other_id: HIM, pseudonym: s.pseudonym, other: { available: true, age: 29, city: 'ouagadougou', verified: true, photo: null }, closed: false }, me: { id: ME } }))
    if (u.pathname === `/v1/conversations/${CONV}/messages`) {
      const at = new Date().toISOString()
      return route.fulfill(json({ items: [
        { id: MINE, conversation_id: CONV, sender_id: ME, mine: true, kind: 'text', state: 'delivered', text: 'Salam', created_at: at },
        { id: HIS, conversation_id: CONV, sender_id: HIM, mine: false, kind: 'text', state: 'delivered', text: 'Bonjour', created_at: at }], has_more: false }))
    }
    if ((u.pathname === '/v1/reports' || u.pathname === '/v1/blocks') && m === 'POST') {
      s.posts.push({ path: u.pathname, body: req.postDataJSON() })
      if (s.mode === 'abort') return route.abort('failed')
      if (s.mode === '429') return route.fulfill(err('RATE_LIMITED', 'Trop de signalements en peu de temps. Réessayez plus tard.', 429))
      if (s.mode === '401') return route.fulfill(err('UNAUTHENTICATED', 'Session expirée', 401))
      if (s.mode === 'slow') await new Promise((r) => setTimeout(r, 1500))
      const b = req.postDataJSON() as any; const now = new Date().toISOString()
      if (u.pathname === '/v1/blocks') return route.fulfill(json({ block: { blocked_id: HIM, created_at: now } }, 201))
      return route.fulfill(json({ report: { id: HIM, target_kind: b.target_kind, reason: b.reason, sla_started_at: now }, case: { id: HIM, sla_started_at: now, sla_state: 'open' }, blocked: !!b.block, sla_hours: 24 }, 201))
    }
    if (u.pathname === '/v1/pin') return route.fulfill(json({ enabled: false, locked: false }))
    return route.fulfill(err('NOT_FOUND', 'Not found', 404))
  })
}
async function open(page: Page, q = '') {
  const target = q.includes('discussion=') ? `/v1/conversations/${CONV}` : `/v1/profiles/${HIM}`
  const loaded = page.waitForResponse((r) => r.url().endsWith(target))
  await page.goto(`/signaler/${HIM}${q}`); await loaded
  await expect(page.locator('#report-block')).toHaveAttribute('aria-busy', 'false')
}
const fresh = (): S => ({ posts: [], mode: 'ok', pseudonym: 'Al-Hassan_BF' })

test.describe('QA 8.5 / 8.6', () => {
  test.skip(!!process.env.E2E_BASE_URL, 'web-only')

  test('widths, screenshots, axe on every state', async ({ page, context, baseURL }) => {
    const s = fresh(); await serve(context, baseURL!, s)
    for (const w of WIDTHS) {
      await page.setViewportSize({ width: w, height: 900 })
      await open(page, `?discussion=${CONV}&message=${HIS}`)
      await expectNoHorizontalOverflow(page)
      await page.screenshot({ path: `${EVID}/screen23-${w}.png`, fullPage: true })
    }
    await open(page)
    await expectAccessible(page)
    await page.getByRole('button', { name: 'Transmettre le signalement' }).click()
    await expect(page.getByRole('alert')).toContainText('Choisissez un motif.')
    await expectAccessible(page)
    await page.screenshot({ path: `${EVID}/screen23-no-reason-error.png`, fullPage: true })
    await page.getByRole('radio').nth(8).check()
    await page.getByRole('button', { name: 'Transmettre le signalement' }).click()
    await expect(page.getByText('Signalement transmis')).toBeVisible()
    await expect(page.getByText('dans les 24 heures')).toBeVisible()
    await expectAccessible(page)
    await page.screenshot({ path: `${EVID}/screen23-sent.png`, fullPage: true })
    expect(s.posts).toHaveLength(1)
    expect(s.posts[0].body).toMatchObject({ target_kind: 'profile', target_id: HIM, reason: 'misrepresented_marital_status', block: true })
  })

  test('block only', async ({ page, context, baseURL }) => {
    const s = fresh(); await serve(context, baseURL!, s); await open(page)
    await page.getByRole('button', { name: 'Bloquer ce profil' }).click()
    await expect(page.getByText('Profil bloqué')).toBeVisible()
    expect(s.posts).toEqual([{ path: '/v1/blocks', body: { blocked_id: HIM } }])
    await expectAccessible(page)
    await page.screenshot({ path: `${EVID}/screen23-blocked.png`, fullPage: true })
  })

  test('failure states keep the input; retry works', async ({ page, context, baseURL }) => {
    const s = fresh(); await serve(context, baseURL!, s); await open(page)
    await page.getByRole('radio').nth(4).check()
    const note = 'Précisions é à ñ 中文 <script>alert(1)</script> \'; DROP TABLE report;--'
    await page.getByRole('textbox').fill(note)
    for (const mode of ['abort', '429'] as const) {
      s.mode = mode
      await page.getByRole('button', { name: 'Transmettre le signalement' }).click()
      await expect(page.getByRole('alert')).toBeVisible()
      await expect(page.getByRole('textbox')).toHaveValue(note)
      await expect(page.getByRole('radio').nth(4)).toBeChecked()
      await expectAccessible(page)
      await page.screenshot({ path: `${EVID}/screen23-error-${mode}.png`, fullPage: true })
    }
    s.mode = '401'
    await page.getByRole('button', { name: 'Transmettre le signalement' }).click()
    await expect(page.getByText('Votre session a expiré. Reconnectez-vous.')).toBeVisible()
    await expect(page.getByRole('link', { name: 'Se reconnecter' })).toBeVisible()
    s.mode = 'ok'; s.posts.length = 0
    await page.getByRole('button', { name: 'Transmettre le signalement' }).click()
    await expect(page.getByText('Signalement transmis')).toBeVisible()
    expect(s.posts[0].body.details).toBe(note)
    expect(await page.evaluate(() => (window as any).__x)).toBeUndefined()
  })

  test('double submit posts once, slow network shows busy, details counter and 1000 cap', async ({ page, context, baseURL }) => {
    const s = fresh(); s.mode = 'slow'; await serve(context, baseURL!, s); await open(page)
    await page.getByRole('radio').nth(0).check()
    const box = page.getByRole('textbox')
    await box.fill('é'.repeat(1005))
    expect((await box.inputValue()).length).toBeLessThanOrEqual(1000)
    const btn = page.getByRole('button', { name: 'Transmettre le signalement' })
    await btn.dblclick()
    await btn.press('Enter').catch(() => {})
    await expect(page.getByText('Signalement transmis')).toBeVisible({ timeout: 8000 })
    expect(s.posts).toHaveLength(1)
  })

  test('XSS pseudonym is escaped; reload and back button; flag only on his bubble', async ({ page, context, baseURL }) => {
    const s = fresh(); s.pseudonym = '<img src=x onerror=window.__x=1>'; await serve(context, baseURL!, s)
    await open(page)
    expect(await page.evaluate(() => (window as any).__x)).toBeUndefined()
    await expect(page.locator('#report-block img[src="x"]')).toHaveCount(0)
    await page.reload(); await expect(page.locator('#report-block')).toHaveAttribute('aria-busy', 'false')
    s.pseudonym = 'Al-Hassan_BF'
    await page.goto(`/discussions/${CONV}`)
    await expect(page.getByRole('link', { name: 'Signaler ce message' })).toHaveCount(1)
    await expect(page.getByRole('link', { name: 'Signaler / Bloquer' })).toBeVisible()
    await page.screenshot({ path: `${EVID}/screen21-thread.png`, fullPage: true })
    await page.getByRole('link', { name: 'Signaler ce message' }).click()
    await expect(page).toHaveURL(new RegExp(`message=${HIS}`))
    await page.goBack(); await expect(page).toHaveURL(new RegExp(`/discussions/${CONV}`))
  })

  test('keyboard-only and 3G-like latency', async ({ page, context, baseURL }) => {
    const s = fresh(); await serve(context, baseURL!, s)
    const cdp = await context.newCDPSession(page)
    await cdp.send('Network.enable')
    await cdp.send('Network.emulateNetworkConditions', { offline: false, latency: 400, downloadThroughput: (400 * 1024) / 8, uploadThroughput: (400 * 1024) / 8 })
    const t0 = Date.now(); await open(page); const loadMs = Date.now() - t0
    test.info().annotations.push({ type: 'load-3g-ms', description: String(loadMs) })
    expect(loadMs).toBeLessThan(8000)
    await page.getByRole('radio').first().focus()
    await page.keyboard.press('ArrowDown'); await page.keyboard.press('ArrowDown')
    await page.keyboard.press('Tab'); await page.keyboard.type('ok')
    await page.keyboard.press('Tab'); await page.keyboard.press('Space')
    await page.keyboard.press('Tab')
    await page.keyboard.press('Enter')
    await expect(page.getByText('Signalement transmis')).toBeVisible()
    await expect(page.getByText('Signalement transmis')).toBeFocused()
    expect(s.posts[0].body).toMatchObject({ reason: 'impersonation', details: 'ok', block: false })
  })
})
