import { launch, EV } from './lib.mjs'
import { readFileSync, writeFileSync } from 'node:fs'
import { devices } from '/home/claude/ankanu-e2e-fix/code/node_modules/playwright-core/index.mjs'
const shim = readFileSync('/tmp/claude-0/-home-claude/6e1b1367-c24b-5f8d-8946-aab9e4dcf519/scratchpad/qa2/tw/cdn-shim.js', 'utf8')
const browser = await launch()
const res = {}
for (const [name, base] of [['head', 'http://localhost:4012'], ['main', 'http://localhost:4014']]) {
  res[name] = []
  for (let i = 0; i < 5; i++) {
    const ctx = await browser.newContext({ ...devices['Pixel 7'], viewport: { width: 360, height: 780 }, locale: 'fr-FR' }); const page = await ctx.newPage()
    await page.route('https://cdn.tailwindcss.com/**', (r) => r.fulfill({ contentType: 'text/javascript', body: shim }))
    await page.route('https://www.google.com/recaptcha/api.js*', (r) => r.fulfill({ contentType: 'text/javascript', body: '' }))
    await page.route('https://fonts.googleapis.com/**', (r) => r.fulfill({ contentType: 'text/css', body: '' }))
    await page.route('https://fonts.gstatic.com/**', (r) => r.abort())
    const sameOrigin = []; page.on('request', (r) => { const u = new URL(r.url()); if (u.origin === base) sameOrigin.push(`${r.method()} ${u.pathname}`) })
    const cdp = await ctx.newCDPSession(page); await cdp.send('Network.enable')
    await cdp.send('Network.emulateNetworkConditions', { offline: false, latency: 150, downloadThroughput: (1.6 * 1024 * 1024) / 8, uploadThroughput: (750 * 1024) / 8 }); await cdp.send('Emulation.setCPUThrottlingRate', { rate: 4 })
    await page.addInitScript(() => { window.__v = { lcp: 0, cls: 0 }; new PerformanceObserver((l) => { for (const e of l.getEntries()) window.__v.lcp = e.startTime }).observe({ type: 'largest-contentful-paint', buffered: true }); new PerformanceObserver((l) => { for (const e of l.getEntries()) if (!e.hadRecentInput) window.__v.cls += e.value }).observe({ type: 'layout-shift', buffered: true }) })
    await page.goto(base + '/auth', { waitUntil: 'load' }); await page.waitForTimeout(1500)
    const m = await page.evaluate(() => { const n = performance.getEntriesByType('navigation')[0]; return { ...window.__v, ttfb: n.responseStart, dcl: n.domContentLoadedEventEnd, load: n.loadEventEnd, docBytes: n.encodedBodySize } })
    res[name].push({ ...m, sameOrigin })
    await ctx.close()
  }
}
const med = (a) => a.sort((x, y) => x - y)[Math.floor(a.length / 2)]
for (const n of ['head', 'main']) console.log(n, 'median LCP', Math.round(med(res[n].map((r) => r.lcp))), 'ms; CLS', med(res[n].map((r) => r.cls)).toFixed(4), '; TTFB', Math.round(med(res[n].map((r) => r.ttfb))), 'ms; load', Math.round(med(res[n].map((r) => r.load))), 'ms; doc bytes', res[n][0].docBytes, '; same-origin requests', JSON.stringify(res[n][0].sameOrigin))
writeFileSync(`${EV}/g7-perf.json`, JSON.stringify(res, null, 2)); await browser.close()
