let cssCache = ''
async function tailwindCss(page: Page): Promise<string> {
  if (cssCache) return cssCache
  const classes = await page.evaluate(() => {
    const set = new Set<string>()
    document.querySelectorAll('[class]').forEach((el) => el.getAttribute('class')!.split(/\s+/).forEach((c) => c && set.add(c)))
    return [...set]
  })
  const compiler = await compile('@import "tailwindcss";', { base: process.cwd(), onDependency: () => {} })
  cssCache = compiler.build(classes)
  return cssCache
}

async function lateCss(context: BrowserContext, css: string, delay: number) {
  await context.route(/^https:\/\/cdn\.tailwindcss\.com\//, (route) =>
    route.fulfill({ status: 200, contentType: 'application/javascript', body: `window.tailwind = { config: {} }; setTimeout(function () { var s = document.createElement('style'); s.id='late-style'; s.textContent = ${JSON.stringify(css)}; document.head.appendChild(s) }, ${delay})` }),
  )
}

function seed() {
  const state = api()
  for (let i = 0; i < 40; i += 1) state.messages.push(text(0x110 + i, FIRST, i % 2 ? ME : HIM, `Message ${i + 1} assez long pour prendre de la place à l'écran, vraiment, encore un peu de texte pour aller à la ligne.`, 39 - i))
  return state
}

  for (const ev of ['message.delivered', 'invite.changed', 'conversation.read']) {
    test(`only ${ev} refreshes the badges on Paramètres`, async ({ page, context, baseURL }) => {
      const state = api()
      state.badges = { invitations: 0, discussions: 0 }
      await serveApi(context, baseURL ?? '', state)
      await page.goto('/settings')
      await page.waitForFunction(() => ((window as unknown as { __fake?: { sockets: unknown[] } }).__fake?.sockets.length ?? 0) === 1)
      state.badges = { invitations: 5, discussions: 7 }
      await fire(page, ev, ev === 'invite.changed' ? { invite_id: id(1) } : { conversation_id: FIRST, account_id: ME, message_id: id(2), sender_id: HIM })
      await expect(page.getByRole('link', { name: /Discussions/ })).toHaveAccessibleName('Discussions 7 discussions non lues')
    })
  }
  for (const [w, h] of [[390, 800], [1280, 900]]) {
    test(`axe with real CSS: thread and list at ${w}`, async ({ browser, baseURL }) => {
      const ctx0 = await browser.newContext({ baseURL })

      await serveApi(ctx0, baseURL ?? '', seed())
      await openThread(p0)
      const css = await tailwindCss(p0)
      await ctx0.close()

      const page = await context.newPage()
      await lateCss(context, css, 0)
      await serveApi(context, baseURL ?? '', seed())
      await openThread(page)
