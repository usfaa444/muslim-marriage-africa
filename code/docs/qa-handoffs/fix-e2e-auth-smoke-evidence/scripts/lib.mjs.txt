export async function newCtx(browser, kind = 'desktop', cookie) {
  const base = kind === 'mobile' ? { ...devices['Pixel 7'], viewport: { width: 360, height: 780 } } : { viewport: { width: 1440, height: 900 } }
  const ctx = await browser.newContext({ ...base, locale: 'fr-FR', timezoneId: 'Africa/Ouagadougou' })
  if (cookie) { const [name, value] = cookie.split('='); await ctx.addCookies([{ name, value, url: BASE }]) }
  return ctx
}
export function watch(page) {
  const w = { gateReads: [], consoleErrors: [], badResponses: [], navs: [], sandboxIgnored: 0, requests: [] }
  const keep = (t) => { if (SANDBOX.some((r) => r.test(t))) { w.sandboxIgnored++; return false } return true }
  page.on('request', (r) => { const u = new URL(r.url()); if (u.origin === BASE) w.requests.push(`${r.method()} ${u.pathname}`); if (u.origin === BASE && u.pathname === '/v1/accounts/gate') w.gateReads.push(`${r.method()} ${r.url()}`) })
  page.on('console', (m) => { if (m.type() === 'error') { const t = `${m.text()} ${m.location().url}`.trim(); if (keep(t)) w.consoleErrors.push(t) } })
  page.on('pageerror', (e) => { if (keep(e.message)) w.consoleErrors.push('pageerror ' + e.message) })
  page.on('response', (r) => { if (r.status() >= 400) { const t = `${r.status()} ${r.request().method()} ${r.url()}`; if (keep(t)) w.badResponses.push(t) } })
  page.on('framenavigated', (f) => { if (f === page.mainFrame()) w.navs.push(f.url().replace(BASE, '')) })
  return w
}
