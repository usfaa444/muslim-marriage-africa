function samplePng(): Buffer {
  const png = new PNG({ width: 8, height: 8 })
  for (let index = 0; index < png.data.length; index += 4) {
    png.data[index] = 40
    png.data[index + 1] = 80
    png.data[index + 2] = 120
    png.data[index + 3] = 255
  }
  return PNG.sync.write(png)
}

  function call(path: string, cookie: string | undefined, method = 'GET', body?: unknown): Promise<Response> {
    const headers: Record<string, string> = {}
    if (cookie) {
      headers.cookie = cookie
    }
    if (body !== undefined) {
      headers['content-type'] = 'application/json'
    }
    const init: RequestInit = { method, headers }
    if (body !== undefined) {
      init.body = JSON.stringify(body)
    }
    return fetch(`${base}${path}`, init)
  }

  async function connected(): Promise<{ brother: Member; sister: Member; conversationId: string }> {

  function connectedSocket(who: Member): Promise<Socket> {
    const socket = connectSocket(base, {
      path: '/v1/realtime',
      addTrailingSlash: false,
      transports: ['websocket'],
      extraHeaders: { cookie: who.cookie },
      reconnection: false,
      forceNew: true,
    })
    sockets.push(socket)
    return new Promise((resolve, reject) => {
      socket.once('connect', () => resolve(socket))
      socket.once('connect_error', (error) => reject(error))
    })
  }

  function noEvent(socket: Socket, name: string, waitMs = 600): Promise<void> {
    return new Promise((resolve, reject) => {
      const onEvent = () => reject(new Error(`unexpected ${name}`))
      socket.once(name, onEvent)
      setTimeout(() => {
        socket.off(name, onEvent)
        resolve()
      }, waitMs)
    })
  }

  it('QA: IDOR — a third member cannot report a message from a conversation she is not in, and blocks are isolated', async () => {

    const sent = await sendText(brother, conversationId, 'Salam')
    const messageId = ((await sent.json()) as any).message.id
    const stranger = await member('sister')
    const theft = await report(stranger, { target_kind: 'message', target_id: messageId, reason: 'harassment' })
    expect(theft.status).toBe(404)
    expect(((await theft.json()) as any).error.message).toBe('Ce message est introuvable.')
    expect(trust.reports).toHaveLength(0)
    // a sister cannot report her own message
    const mine = await sendText(sister, conversationId, 'Wa alaykum')
    const mineId = ((await mine.json()) as any).message.id
    expect((await report(sister, { target_kind: 'message', target_id: mineId, reason: 'harassment' })).status).toBe(404)
    // block isolation: stranger still sees the brother after sister blocks him
    expect((await call('/v1/blocks', sister.cookie, 'POST', { blocked_id: brother.id })).status).toBe(201)
    expect(await browseIds(stranger)).toContain(brother.id)
    expect((await call(`/v1/profiles/${brother.id}`, stranger.cookie)).status).toBe(200)
    expect((await call(`/v1/profiles/${brother.id}`, sister.cookie)).status).toBe(404)
  })

  it('QA: a message Report never stores the message text', async () => {

  it('QA: AC 8.5-2 SLA boundary exactly at 24 h, and the breach follows a changed report_sla_hours', async () => {
    const { brother, sister } = await connected()
    expect((await report(sister, { target_kind: 'profile', target_id: brother.id, reason: 'hateful' })).status).toBe(201)
    const staff = await openTestStaff(base, staffStore, 'operator')
    const metrics = async (at: Date) => {
      now = new Date(Math.max(now.getTime() + 31_000, at.getTime() - 120_000))
      const cookie = await signInTestStaff(base, staffStore, staff)
      now = at
      return staffMetrics(cookie)
    }
    const start = new Date(trust.cases[0]!.sla_started_at).getTime()
    expect((await metrics(new Date(start + 24 * 3600_000))).reports).toMatchObject({ open: 1, sla_breach: 0, sla_hours: 24 })
    expect((await metrics(new Date(start + 24 * 3600_000 + 1000))).reports).toMatchObject({ open: 1, sla_breach: 1 })
    slaHours = 6
    expect((await metrics(new Date(start + 25 * 3600_000))).reports).toMatchObject({ open: 1, sla_breach: 1, sla_hours: 6 })
    slaHours = null
    now = new Date(now.getTime() + 31_000)
    expect((await call('/v1/staff/metrics', await signInTestStaff(base, staffStore, staff))).status).toBe(503)
    slaHours = 24
    trust.cases[0]!.first_human_at = new Date(start + 48 * 3600_000).toISOString()
    expect((await metrics(new Date(start + 72 * 3600_000))).reports).toMatchObject({ open: 0, sla_breach: 0, oldest_open_started_at: null })
    expect(caseSlaState(trust.cases[0]!, now, 24)).toBe('first_touch_late')
  })

  it('QA: rate limit refuses the 11th Report in an hour and writes nothing; the window slides', async () => {
    const { brother, sister } = await connected()
    reportRate = 3
    for (let i = 0; i < 3; i += 1) {
      now = new Date(now.getTime() + 60_000)
      expect((await report(sister, { target_kind: 'profile', target_id: brother.id, reason: 'indecency' })).status).toBe(201)
    }
    now = new Date(now.getTime() + 60_000)
    const limited = await report(sister, { target_kind: 'profile', target_id: brother.id, reason: 'indecency' })
    expect(limited.status).toBe(429)
    const body = (await limited.json()) as any
    expect(body.error.code).toBe('RATE_LIMITED')
    expect(body.error.retryable).toBe(true)
    expect(trust.reports).toHaveLength(3)
    now = new Date(now.getTime() + 61 * 60_000)
    expect((await report(sister, { target_kind: 'profile', target_id: brother.id, reason: 'indecency' })).status).toBe(201)
    reportRate = null
    expect((await report(sister, { target_kind: 'profile', target_id: brother.id, reason: 'indecency' })).status).toBe(503)
    expect(trust.reports).toHaveLength(4)
  })

  it('QA: injection-shaped and odd bodies are refused or stored verbatim, never executed', async () => {
    const { brother, sister } = await connected()
    const bad: unknown[] = [
      null,
      'string-body',
      { target_kind: 'profile', target_id: brother.id },
      { target_kind: 'profile', target_id: `${brother.id}' OR 1=1--`, reason: 'indecency' },
      { target_kind: 'profile', target_id: brother.id, reason: "indecency'; DROP TABLE report;--" },
      { target_kind: 'PROFILE', target_id: brother.id, reason: 'indecency' },
      { target_kind: 'profile', target_id: brother.id, reason: 'indecency', block: 'true' },
      { target_kind: 'profile', target_id: brother.id, reason: 'indecency', details: 42 },
      { target_kind: 'profile', target_id: sister.id, reason: 'indecency' },
    ]
    for (const body of bad) {
      const response = await report(sister, body)
      expect([400, 404]).toContain(response.status)
    }
    expect(trust.reports).toHaveLength(0)
    expect(trust.cases).toHaveLength(0)
  })

  it('QA: AC 8.6 — Block closes browse, profile, favourites, invites and the thread both ways', async () => {
    const { brother, sister, conversationId } = await connected()
    expect((await sendText(brother, conversationId, 'Salam')).status).toBe(201)
    expect((await call('/v1/favourites', sister.cookie, 'POST', { target_id: brother.id })).status).toBe(201)
    expect((await call('/v1/blocks', sister.cookie, 'POST', { blocked_id: brother.id })).status).toBe(201)
    expect((await call('/v1/blocks', sister.cookie, 'POST', { blocked_id: brother.id })).status).toBe(200)
    expect(trust.blocks).toHaveLength(1)
    expect(await browseIds(brother)).not.toContain(sister.id)
    expect(await browseIds(sister)).not.toContain(brother.id)
    expect((await call(`/v1/profiles/${sister.id}`, brother.cookie)).status).toBe(404)
    const favouriteList = await call('/v1/favourites', sister.cookie)
    expect(JSON.stringify(await favouriteList.json())).not.toContain(brother.id)
    const invite = await call('/v1/invites', brother.cookie, 'POST', { to_id: sister.id })
    expect(invite.status).toBe(404)
    const send = await sendText(brother, conversationId, 'encore')
    expect(send.status).toBe(409)
    expect(((await send.json()) as any).error.details?.reason).toBe('conversation_closed')
    const sendBack = await sendText(sister, conversationId, 'encore')
    expect(sendBack.status).toBe(409)
    const thread = await call(`/v1/conversations/${conversationId}`, brother.cookie)
    expect(thread.status).toBe(200)
    const threadBody = (await thread.json()) as any
    expect(threadBody.conversation.closed).toBe(true)
    expect(threadBody.conversation.other.available).toBe(false)
  })
