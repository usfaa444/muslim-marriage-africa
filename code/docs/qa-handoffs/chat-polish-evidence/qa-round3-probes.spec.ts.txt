const OUT = '/tmp/claude-0/q3/probe-out.txt'
function seed(n = 40) {
  const state = api()
  for (let i = 0; i < n; i += 1) state.messages.push(text(0x110 + i, FIRST, i % 2 ? ME : HIM, `Message ${i + 1} assez long pour prendre de la place à l'écran, vraiment, encore un peu de texte pour aller à la ligne.`, 39 - i))
  return state
}

test.describe('QA round 3: stale Paramètres windows', () => {
  test.skip(!!process.env.E2E_BASE_URL, STUB_SKIP)
  const sw = (p: import('@playwright/test').Page) => p.getByRole('switch', { name: 'Confirmations de lecture' })

  test('two windows: live event, focus, stale click, overtaken read', async ({ context, baseURL }) => {
    const state = api()
    await serveApi(context, baseURL ?? '', state)
    const a = await context.newPage(); const b = await context.newPage()
    await a.goto('/settings'); await b.goto('/settings')
    await expect(sw(a)).toHaveAttribute('aria-checked', 'true'); await expect(sw(b)).toHaveAttribute('aria-checked', 'true')
    // A turns off; B (no event, hidden) stays stale
    await sw(a).click()
    await expect(sw(a)).toHaveAttribute('aria-checked', 'false')
    expect(state.setting).toBe(false)
    await expect(sw(b)).toHaveAttribute('aria-checked', 'true') // stale, as expected without a signal
    // live event reaches B
    await realtimeReady(b).catch(() => {})
    await fire(b, 'chat.receipts', { account_id: ME })
    await expect(sw(b)).toHaveAttribute('aria-checked', 'false')
    // now server flips back from elsewhere; B gets only a focus
    state.setting = true
    await b.evaluate(() => window.dispatchEvent(new Event('focus')))
    await expect(sw(b)).toHaveAttribute('aria-checked', 'true')
    // stale click: server false, B shows true (no signal), member clicks: asks for off
    state.setting = false
    state.settingPuts.length = 0
    await sw(b).click()
    await expect.poll(() => state.settingPuts).toEqual([{ read_receipts: false }])
    await expect(sw(b)).toHaveAttribute('aria-checked', 'false')
    // overtaken read: GET delayed with the old value; click in between
    state.setting = true
    await b.evaluate(() => window.dispatchEvent(new Event('focus')))
    await expect(sw(b)).toHaveAttribute('aria-checked', 'true')
    let release: () => void = () => {}
    const gate = new Promise<void>((r) => { release = r })
    await b.route(/\/v1\/chat\/settings$/, async (route) => {
      if (route.request().method() === 'GET') { const snapshot = state.setting; await gate; return route.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ read_receipts: snapshot }) }) }
      return route.fallback()
    })
    await b.evaluate(() => window.dispatchEvent(new Event('focus'))) // GET in flight, old value true
    state.settingPuts.length = 0
    await sw(b).click() // asks false
    await expect.poll(() => state.settingPuts).toEqual([{ read_receipts: false }])
    release()
    await b.waitForTimeout(500)
    await expect(sw(b)).toHaveAttribute('aria-checked', 'false')
    // rapid double click does not double-PUT, keeps focus; keyboard
    await b.unroute(/\/v1\/chat\/settings$/)
    await sw(b).focus()
    state.settingPuts.length = 0
    await b.keyboard.press('Space')
    await expect(sw(b)).not.toHaveAttribute('aria-busy', 'true')
    await expect(sw(b)).toBeFocused()
    await b.keyboard.press('Space')
    await expect.poll(() => state.settingPuts.length).toBe(2)
    await expect(sw(b)).toBeFocused()
    // PUT failure: reverts + error
    await b.route(/\/v1\/chat\/settings$/, (route) => route.request().method() === 'PUT' ? route.fulfill({ status: 500, contentType: 'application/json', body: '{"error":{"code":"X","message":"x","details":null}}' }) : route.fallback())
    const before = await sw(b).getAttribute('aria-checked')
    await sw(b).click()
    await expect(sw(b)).toHaveAttribute('aria-checked', before!)
    await expect(b.locator('#receipts-error')).toBeVisible()
  })
