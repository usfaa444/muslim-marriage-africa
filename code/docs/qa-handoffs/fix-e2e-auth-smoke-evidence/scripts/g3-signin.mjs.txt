const browser = await launch()
const STUB = `window.grecaptcha={_cb:null,render:function(el,o){this._cb=o.callback;setTimeout(function(){o.callback()},20);return 0},getResponse:function(){return 'test-ok'},reset:function(){}};setTimeout(function(){},0)`
async function stubGoogle(page) {
  await page.route('https://www.google.com/recaptcha/api.js*', (r) => r.fulfill({ contentType: 'text/javascript', body: `${STUB};if(window.onRecaptchaLoad)window.onRecaptchaLoad()` }))
  await page.route('https://cdn.tailwindcss.com/**', (r) => r.fulfill({ contentType: 'text/javascript', body: 'window.tailwind={}' }))
}
async function fill(page, email, pw) {
  await page.locator('#login-email').fill(email)
  await page.locator('#login-password').fill(pw)
  await page.waitForFunction(() => !document.getElementById('login-submit').disabled, null, { timeout: 5000 })
}

// T07 successful sign-in for members at different steps (UI -> POST /v1/sessions -> gate -> step)
for (const [label, want] of [['a', '/email-verification'], ['b', '/phone'], ['c', '/otp'], ['d', '/id-liveness'], ['e', '/onboarding']]) {
  const ctx = await newCtx(browser, 'desktop'); const page = await ctx.newPage(); const w = watch(page); await stubGoogle(page)
  const posts = []; page.on('response', (r) => { if (r.url().endsWith('/v1/sessions') && r.request().method() === 'POST') posts.push(r.status()) })
  await page.goto(BASE + '/auth'); await page.waitForLoadState('networkidle')
  const gateBefore = w.gateReads.length
  await fill(page, accounts[label].email, PW)
  await page.locator('#login-submit').click()
  await page.waitForURL((u) => u.pathname !== '/auth', { timeout: 10000 }).catch(() => {})
  await page.waitForLoadState('networkidle')
  record(`T07-${label}`, `UI sign-in (member at ${accounts[label].gate.step}) -> POST /v1/sessions 201 -> exactly one gate read after submit -> ${want}`, posts.join() === '201' && gateBefore === 0 && w.gateReads.length === 1 && new URL(page.url()).pathname === want, { posts, gateBefore, gateReadsTotal: w.gateReads.length, url: page.url().replace(BASE, ''), consoleErrors: w.consoleErrors, badResponses: w.badResponses })
  await ctx.close()
}
