// QA sandbox stub for https://www.google.com/recaptcha/api/siteverify (Google is blocked here).
// Mimics Google's documented behaviour for the public TEST secret: any token succeeds.
// Any other secret is refused, so a misconfigured api cannot pass.
import { createServer } from 'node:https'
import { readFileSync } from 'node:fs'
const TEST_SECRET = '6LeIxAcTAAAAAGG-vFI1TnRWxMZNFuojJ4WifJWe'
createServer({ key: readFileSync('/stub/google.key'), cert: readFileSync('/stub/google.crt') }, (req, res) => {
  let body = ''
  req.on('data', (c) => (body += c))
  req.on('end', () => {
    const p = new URLSearchParams(body)
    console.log(JSON.stringify({ method: req.method, url: req.url, secretIsTest: p.get('secret') === TEST_SECRET, token: p.get('response') }))
    res.writeHead(200, { 'content-type': 'application/json' })
    res.end(JSON.stringify(req.url.startsWith('/recaptcha/api/siteverify') && p.get('secret') === TEST_SECRET ? { success: true } : { success: false, 'error-codes': ['invalid-input-secret'] }))
  })
}).listen(443, '0.0.0.0', () => console.log('siteverify stub on 443'))
